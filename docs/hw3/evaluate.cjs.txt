function loadSource(root, relative, cache = new Map()) {
  let filename = path.resolve(root, relative);
  if (!path.extname(filename)) filename += fs.existsSync(filename + '.ts') ? '.ts' : '.tsx';
  if (cache.has(filename)) return cache.get(filename).exports;
  const module = { exports: {} };
  cache.set(filename, module);
  const code = ts.transpileModule(fs.readFileSync(filename, 'utf8'), {
    compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2022, jsx: ts.JsxEmit.ReactJSX, esModuleInterop: true },
    fileName: filename,
  }).outputText;
  const customRequire = (id) => {
    if (id.startsWith('@/')) return loadSource(root, 'src/' + id.slice(2), cache);
    if (id.startsWith('.')) return loadSource(root, path.resolve(path.dirname(filename), id), cache);
    // Both implementations use the same pinned React version; one instance avoids duplicate-hook errors.
    return bRequire(id);
  };
  new Function('exports', 'require', 'module', code)(module.exports, customRequire, module);
  return module.exports;
}

async function review(run) {
  const root = path.join(base, 'b4gone-hw3-' + run.toLowerCase());
  const cache = new Map();
  const isA = run === 'A';
  const module = loadSource(root, isA ? 'src/lib/purchase.ts' : 'src/features/items/item-schema.ts', cache);
  const results = cases.map(([name, patch, expected, fields]) => {
    const draft = { ...basic, ...patch };
    const raw = isA ? module.validateNewPurchase({ itemName: draft.name, ...Object.fromEntries(Object.entries(draft).filter(([key]) => key !== 'name')) }) : module.validateItemDraft(draft, TODAY);
    const accepted = isA ? raw.ok : raw.success;
    const value = isA ? (raw.value && { ...raw.value, name: raw.value.itemName }) : raw.data;
    const pass = accepted === expected && (!fields || Object.entries(fields).every(([key, val]) => value?.[key] === val));
    return { name, pass, expectedAccepted: expected, actualAccepted: accepted };
  });
  const Form = loadSource(root, isA ? 'src/components/NewPurchaseForm.tsx' : 'src/features/items/AddItemForm.tsx', cache)[isA ? 'NewPurchaseForm' : 'AddItemForm'];
  const fieldName = isA ? 'itemName' : 'name';
  for (const mode of ['pending', 'rejection', 'resolution']) {
    let resolve, reject;
    const promise = new Promise((yes, no) => { resolve = yes; reject = no; });
    promise.catch(() => {});
    let calls = 0;
    const onSubmit = () => { calls++; return promise; };
    const ui = render(React.createElement(Form, isA ? { onSubmit } : { today: TODAY, onSubmit }));
    const name = ui.container.querySelector('input[name="' + fieldName + '"]');
    const purchase = ui.container.querySelector('input[name="purchaseDate"]');
    fireEvent.change(name, { target: { value: 'Desk lamp' } });
    fireEvent.change(purchase, { target: { value: '2026-10-20' } });
    const form = ui.container.querySelector('form');
    fireEvent.submit(form);
    if (mode === 'pending') {
      // Re-enter the same draft if an implementation cleared it too early, then try a second submit.
      const preserved = name.value === 'Desk lamp';
      if (!name.value) {
        fireEvent.change(name, { target: { value: 'Desk lamp' } });
        fireEvent.change(purchase, { target: { value: '2026-10-20' } });
      }
      fireEvent.submit(form);
      results.push({ name: 'draft retained while submission is pending', pass: preserved });
      results.push({ name: 'duplicate submission blocked while pending', pass: calls === 1, callbackCalls: calls });
      await act(async () => { resolve(); await promise; });
    } else if (mode === 'rejection') {
      await act(async () => { reject(new Error('Simulated save failure')); await promise.catch(() => {}); });
      results.push({ name: 'failed async submission retains draft and shows retry error', pass: name.value === 'Desk lamp' && /try again|retry/i.test(ui.container.textContent) });
    } else {
      await act(async () => { resolve(); await promise; });
      results.push({ name: 'successful submission clears draft', pass: name.value === '' && purchase.value === '' });
    }
    cleanup();
  }
  return { run, fixedToday: TODAY, passed: results.filter(x => x.pass).length, total: results.length, results };
}

(async () => {
  const result = { method: 'One post-generation review checklist based on the existing PRD; independent of both generated test suites. Original source files loaded read-only via TypeScript transpilation; no application source changes.', reviews: [await review('A'), await review('B')] };
  const destination = path.join(__dirname, 'results.json');
  fs.writeFileSync(destination, JSON.stringify(result, null, 2) + '\n');
  for (const run of result.reviews) console.log(JSON.stringify({ run: run.run, passed: run.passed, total: run.total, failures: run.results.filter(x => !x.pass) }, null, 2));
})().catch(error => { console.error(error); process.exitCode = 1; });
